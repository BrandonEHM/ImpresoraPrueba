  nombre: string;
  totalVisitantes: number;
  precio: string;
  fechaHora: string;
  lugar: string;
  boletosSeleccionados: string;

}

interface DatosQR {
  totalVisitantes: number;
  boletos: string;
  precioTotal: string;
  'fecha-expiracion': string;
}


@Injectable({
  providedIn: 'root',
})
export class Printing {

  constructor() { }
  //obetener hora y fecha actual

    const ahora = new Date();
    return formatDate(ahora, 'dd/MM/yyyy HH:mm:ss', 'es-MX');
  }



  //Obtener datos para el ticket
  private obtenerDatosTicket(): DatosTicket | null {
    const nombre = nombreVisitante;
    const totalVisitantes = ExportTotalVisitantes;
    const precio = PrecioTotal.toFixed(2);
    const fechaHora = this.obtenerFechaActual();
    const lugar = 'Zacatecas, México';
    const boletosSeleccionados: string = boletosselect;


    if (!nombre || !totalVisitantes || !precio) {
      console.error('Faltan datos para generar el ticket');
      alert('Faltan datos para generar el ticket');
      return null;
    }

    const fechaExpiracion = new Date(Date.now() + 2 * 24 * 60 * 60 * 1000);
    return {
      fechaHora,

      nombre,

      boletosSeleccionados
    };
  }


  // Genera los datos para el código QR

  private generarDatosQR(datosTicket: DatosTicket): DatosQR {
    const fechaExpiracion = new Date(Date.now() + 2 * 24 * 60 * 60 * 1000);

    return {

      totalVisitantes: datosTicket.totalVisitantes,
      boletos: datosTicket.boletosSeleccionados,
      precioTotal: datosTicket.precio,
      'fecha-expiracion': fechaExpiracion.toLocaleString('es-MX')
    };
  }


  //Convierte el nivel de corrección QR a código ESC/POS
  //ISO 18004 define 4 niveles: L(7%), M(15%), Q(25%), H(30%)


  private obtenerCodigoECC(nivel: NivelCorreccionQR): string {
    const codigosECC: Record<NivelCorreccionQR, string> = {
      'L': '\x30', // 7% recuperación
      'M': '\x31', // 15% recuperación
      'Q': '\x32', // 25% recuperación
      'H': '\x33'  // 30% recuperación
    };
    return codigosECC[nivel];
  }

  //Generar datos para impresión comandos ESC/POS 
  private generarComandosESCPOS(datosTicket: DatosTicket,
    datosQR: DatosQR,
    nivelCorreccion: NivelCorreccionQR = 'M'): string[] {

    const datosQRString = JSON.stringify(datosQR);
    const qrDatosLength = datosQRString.length;
    const commandLength = qrDatosLength + 3;
    const eccCode = this.obtenerCodigoECC(nivelCorreccion);

    const comandos: string[] = [
      // Inicializar impresora
      '\x1B\x40',

      // Configurar fuente y estilo
      // Alineacion centrada
      '\x1B\x61\x01',

      // Encabezado
      '\x1B\x21\x30',      // ESC ! 48 - Texto doble alto y doble ancho
      '\x1B\x21\x08',      // ESC ! 8 - Enfatizado
      'Museo\n',
      '\x1B\x21\x00',      // ESC ! 0 - Texto normal
      '\n',

      // Lugar
      '\x1B\x21\x10',
      `Lugar: ${datosTicket.lugar}\n`,
      '\x1B\x21\x00',
      '\n',

      // Fecha y hora
      `Emitido: ${datosTicket.fechaHora}\n`,
      '\n',

      // Nombre del visitante
      '\x1B\x21\x08',
      'Bienvenido(a):\n',
      '\x1B\x21\x00',
      `${datosTicket.nombre}\n`,

      //Total visitantes
      '\x1B\x21\x08',
      `Total visitantes: ${datosTicket.totalVisitantes}\n`,
      '\x1B\x21\x00',
      '\n',

      // Precio total
      '\x1B\x21\x18',
      `Precio total: $${datosTicket.precio}\n`,
      '\x1B\x21\x00',
      '\n',

      // Instrucciones para el código QR
      // Configuración del código QR (ISO 18004)
      '\x1D\x28\x6B\x04\x00\x31\x41\x32\x00',  // GS ( k - Modelo 2 (QR Code)
      '\x1D\x28\x6B\x03\x00\x31\x43\x08',      // Tamaño del módulo: 8 unidades
      `\x1D\x28\x6B\x03\x00\x31\x45${eccCode}`, // Nivel de corrección según ISO 18004

      // Almacenar datos del QR
      `\x1D\x28\x6B${String.fromCharCode(commandLength & 0xFF)}${String.fromCharCode((commandLength >> 8) & 0xFF)}\x31\x50\x30${datosQRString}`,

      '\x1D\x28\x6B\x03\x00\x31\x51\x30',      // Imprimir el QR

      '\x1B\x61\x00',      // Alineación izquierda
      '\n\n\n\n\n',       // Avanzar papel
      '\x1D\x56\x41\x10'   // Cortar papel
    ];

    return comandos;
  }

  //Convertir ESC/POS en formato imprimible creando
  // iframe oculto con los datos en formato RAW
  private imprimirComandosRAW(comandos: string[]): void {
    const rawData = comandos.join('');
    //Crear iframe oculto
    const iframe = document.createElement('iframe');
    iframe.style.position = 'fixed';
    iframe.style.width = '0';
    iframe.style.height = '0';
    iframe.style.border = 'none';
    document.body.appendChild(iframe);

    document.body.appendChild(iframe);
    const doc = iframe.contentWindow?.document;
    if (!doc) {
      console.error('No se pudo acceder al documento del iframe para imprimir.');
      document.body.removeChild(iframe);
      return;
    }
    //Escribir datos RAW en el iframe
    doc.open();
    doc.write(`
        <head>
          <style>
            @page {
              size: 80mm auto;
              margin: 0;
            }
            body {
              margin: 0;
              padding: 0;
              font-family: monospace;
              white-space: pre-wrap;
            }
          </style>
        </head>
        <body>${this.escaparHTML(rawData)}</body>
      </html>
    `);
    doc.close();

    // Esperar a que carge y luego imprimir
    iframe.onload = () => {
      setTimeout(() => {
        iframe.contentWindow?.print();

        // Limpiar después de imprimir
        setTimeout(() => {
          document.body.removeChild(iframe);
        }, 1000);
      }, 250);
    };
  }
  //Escapar caracteres especiales para HTML
  private escaparHTML(texto: string): string {
    const map: { [key: string]: string } = {
      '&': '&amp;',
      '<': '&lt;',
      '>': '&gt;',
      '"': '&quot;',
      "'": '&#039;',
    };
    return texto.replace(/[&<>"']/g, function (m) { return map[m]; });
  }

  //Método principal para imprimir el ticket

  public imprimirTicket(): void {
    const datosTicket = this.obtenerDatosTicket();
    if (!datosTicket) {
      console.error('No se pudieron obtener los datos del ticket');
      return;
    }
    // Generar datos del QR
    const datosQR = this.generarDatosQR(datosTicket);
    // Generar comandos ESC/POS
    const comandos = this.generarComandosESCPOS(datosTicket, datosQR, nivelErrorQR);
    this.imprimirComandosRAW(comandos);
    console.log('Ticket enviado a impresión', {
      datosTicket,
      datosQR,
      nivelErrorQR
    });
  }
